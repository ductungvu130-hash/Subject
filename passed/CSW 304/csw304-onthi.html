📏 Nguyên lý: Container → Row → Col

<div class="container">       <!-- giới hạn độ rộng + căn giữa -->
  <div class="row">            <!-- hàng ngang, dùng flexbox -->
    <div class="col-md-4">...</div>   <!-- 4/12 = 1/3 màn hình -->
    <div class="col-md-8">...</div>   <!-- 8/12 = 2/3 màn hình -->
  </div>
</div>
ClassÝ nghĩaDùng khi nào
col-22/12 cột, luôn luônLayout cố định
col-md-33/12 từ màn ≥768pxResponsive tablet
col-lg-44/12 từ màn ≥992pxDesktop
colChia đều tự độngKhi muốn bằng nhau
⚠ Quy tắc tổng = 12: Các col-* trong cùng 1 row phải cộng lại = 12.

jQuery

9 · jQuery là gì? Tại sao dùng?

💡 jQuery = viết ít, làm nhiều

jQuery là thư viện JavaScript giúp:

🎯
Chọn phần tử
$('#id'), $('.class')
🖱️
Xử lý sự kiện
.click(), .on()
🔄
Gọi API
$.get(), $.post()
🏗️
Thao tác DOM
.html(), .append()
So sánh JS thuần vs jQuery
// ── JavaScript thuần ──
document.getElementById('btn').addEventListener('click', function() {
  document.getElementById('box').innerHTML = 'Hello!';
});

// ── jQuery (ngắn hơn rất nhiều) ──
$('#btn').click(function() {
  $('#box').html('Hello!');
});

⏰ Chờ trang load xong mới chạy jQuery

// CÁC CÁCH VIẾT TƯƠNG ĐƯƠNG – đều đúng
$(document).ready(function() {
  // Mọi code jQuery viết trong đây
});

// Viết tắt phổ biến nhất:
$(function() {
  // Mọi code jQuery viết trong đây
});
💡 Bắt buộc: Toàn bộ code jQuery phải nằm trong $(function(){...}). Nếu không, jQuery sẽ chạy trước khi HTML được tải và sẽ không tìm thấy phần tử nào!

13 · $.each() – Duyệt mảng dữ liệu từ API

🔄 Cách duyệt và render dữ liệu

// Giả sử data là mảng trả về từ API:
var products = [
  { id: 1, name: "Hoa hồng", price: 250000, image: "hoa1.jpg" },
  { id: 2, name: "Hoa cúc",  price: 180000, image: "hoa2.jpg" }

// Cách 1: $.each(array, function(index, item))
$.each(products, function(i, product) {
  var html = `
    <div class="col">
      <div class="card product-card">
        <img src="${product.image}" class="card-img-top"/>

          </button>
        </div>
      </div>
    </div>
  `;
  $('#product-list').append(html);
});

// Cách 2: for...of (ES6, cũng được dùng)
for (var product of products) {

$.get / $.post – Quan trọng nhất!

14 · Gọi API với $.get() / $.post() – KHÔNG dùng $.ajax()

⭐ Đây là phần xuất hiện trong TẤT CẢ đề thi CSW 304 Quarter 1 & 2!
$.get() và $.post() là 2 hàm "rút gọn" (shorthand) của $.ajax(), giúp viết code ngắn hơn và dễ đọc hơn.

📐 Cú pháp chung

// CÚ PHÁP $.get():
$.get(url, data, successCallback, dataType);

// CÚ PHÁP $.post():
$.post(url, data, successCallback, dataType);

//  url             → địa chỉ API
//  data            → tham số gửi kèm (object {key: value})
//  successCallback → function(response) chạy khi nhận được dữ liệu
//  dataType        → kiểu dữ liệu trả về, thường là 'json'

🟢 GET Request – Lấy dữ liệu

$.get(
  'https://api.example.com/data',   // URL
  { shoppingCart: 'All' },          // Parameters (?shoppingCart=All)
  function(data) {                  // Callback khi thành công
    console.log(data);
    renderCart(data);
  },
  'json'                            // Dữ liệu trả về dạng JSON
);

function renderCart(response) {
  $('#cart-list').empty();          // xóa dữ liệu cũ trước khi render
  $.each(response, function(i, item) {
    $('#cart-list').append(`
      <tr>
        <td>${item.name}</td>
        <td>${item.price}</td>
        <td><button class="btn-remove" data-id="${item.id}">✕</button></td>
      </tr>
    `);
  });
}

🔵 POST Request – Gửi dữ liệu lên server

$('#btnOrder').click(function() {

  // 1. Lấy giá trị form
  var fullName     = $('#fullName').val().trim();
  var phoneNumber  = $('#phone').val().trim();
  var address      = $('#address').val().trim();
  var note         = $('#note').val().trim();

  // 2. Validate (kiểm tra bắt buộc theo đề)
  if (!fullName || !phoneNumber || !address) {
    alert('Vui lòng điền đầy đủ thông tin bắt buộc!');
    return; // DỪNG lại, không gửi
  }

  // 3. Gửi POST request bằng $.post()
  $.post(
    'URL_TU_MOODLE',
    {
      fullName:    fullName,
      phoneNumber: phoneNumber,
      address:     address,
      note:        note
    },
    function(response) {
      // Đề yêu cầu: alert response từ server
      alert(response.message || JSON.stringify(response));
    },
    'json'
  );

});

⏰ Gọi API ngay khi trang load (giống script.js mẫu)

$(document).ready(function () {

  let products = [];

  // Gọi API ngay khi trang load xong
  $.get(

            </div>
          </div>
        </div>
      `);
    }
  }

});

🔴 Validation nâng cao – Red border (Đề 3 – Room)

📋 Yêu cầu đề: "If a validation error occurs, a red border should be displayed around the invalid field along with the message: 'Please fill out this field.'" 
$(function() {

  $('#btnAddRoom').click(function() {
    var isValid = true;

    // Reset trạng thái trước
    $('.form-control').removeClass('is-invalid');
    $('.invalid-feedback').remove();

    // Kiểm tra Room Number
    var roomNumber = $('#roomNumber').val().trim();
    if (!roomNumber) {
      showError('#roomNumber', 'Please fill out this field.');
      isValid = false;
    }

    // Kiểm tra Price: không được rỗng VÀ phải là số
    var price = $('#price').val().trim();
    if (!price) {
      showError('#price', 'Please fill out this field.');
      isValid = false;
    } else if (isNaN(price) || Number(price) <= 0) {
      showError('#price', 'Price must be a valid number.');
      isValid = false;
    }

    if (isValid) {
      // Gửi form
      submitRoom(roomNumber, price);
    }
  });

  // Helper function thêm viền đỏ + message
  function showError(selector, message) {
    $(selector)
      .addClass('is-invalid')         // Bootstrap class → viền đỏ
      .after(`<div class="invalid-feedback">${message}</div>`);
  }

  // Xóa lỗi khi người dùng gõ lại
  $(document).on('input', '.form-control', function() {
    $(this).removeClass('is-invalid');
    $(this).next('.invalid-feedback').remove();
  });

});

16 · Đề 1 – Giỏ hàng (Shopping Cart)

📋 Yêu cầu đề Quarter 1
a) GET request với param shoppingCart=All → hiển thị vào "Thông tin giỏ hàng"
b) Nút remove (✕) → xóa sản phẩm khỏi giỏ
c) Form đặt hàng → POST request → validate 3 trường bắt buộc

✅ Code đầy đủ Question 2 – Shopping Cart

$(document).ready(function () {

  let cartItems = [];

  // ════ a) Lấy danh sách giỏ hàng khi trang load ════
  $.get(
    'URL_TU_MOODLE',
    { shoppingCart: 'All' },
    function (data) {
      cartItems = data;
      renderCart(cartItems);
    },
    'json'
  );

  // Render danh sách giỏ hàng vào bảng
  function renderCart(items) {
    $('#cartList').empty();

    if (items.length === 0) {
      $('#cartList').html('<tr><td colspan="5" class="text-center">Giỏ hàng trống</td></tr>');
      return;
    }

    for (let item of items) {
      $('#cartList').append(`
        <tr data-id="${item.id}">
          <td><img src="${item.image}" width="60"/></td>
          <td>${item.name}</td>
          <td>${item.quantity}</td>
          <td>${parseInt(item.price).toLocaleString()}đ</td>
          <td>
            <button class="btn btn-sm btn-danger btn-remove" data-id="${item.id}">✕</button>
          </td>
        </tr>
      `);
    }
  }

  // ════ b) Xóa sản phẩm ════
  // Phải dùng $('#cartList').on() vì button được tạo DYNAMICALLY (giống script.js mẫu)
  $('#cartList').on('click', '.btn-remove', function () {
    let _id = $(this).attr('data-id');
    for (let i in cartItems) {
      if (cartItems[i].id == _id) {
        cartItems.splice(i, 1);
        break;
      }
    }
    renderCart(cartItems);
  });

  // ════ c) Đặt hàng – POST bằng $.post() ════
  $('#btnOrder').click(function () {
    let fullName    = $('#fullName').val().trim();
    let phoneNumber = $('#phone').val().trim();
    let address     = $('#address').val().trim();
    let note        = $('#note').val().trim();

    // Validate 3 trường bắt buộc
    if (!fullName || !phoneNumber || !address) {
      alert('⚠ Vui lòng điền đầy đủ: Họ và tên, Số điện thoại, Địa chỉ!');
      return;
    }

    $.post(
      'URL_TU_MOODLE',
      { fullName, phoneNumber, address, note },
      function (response) {
        alert(response.message || 'Đặt hàng thành công!');
      },
      'json'
    );
  });

});

$(document).ready(function () {

18 · Đề 3 – Quản lý phòng (Rooms)

📋 Yêu cầu đề
a) GET GetRooms=All và RoomType → hiển thị room list (sort ASC by price) + room types
b) Lọc phòng theo loại (filter by type)
c) Thêm phòng mới với validation (Room Number không rỗng, Price là số)
d) Filter theo giá: <$15 / $15–$20 / >$20

✅ Code đầy đủ Question 2 – Rooms

$(function() {

  var allRooms     = [];
  var allRoomTypes = [];

  // ════ a) Load dữ liệu khi trang load ════
  // Gọi 2 API song song
  $.when(
    $.ajax({ url: 'URL_TU_MOODLE', method: 'GET', data: { GetRooms: 'All' } }),
    $.ajax({ url: 'URL_TU_MOODLE', method: 'GET', data: { RoomType: 'All' } })
  ).then(function(roomsRes, typesRes) {
    allRooms     = roomsRes[0];
    allRoomTypes = typesRes[0];
    renderRoomTypes(allRoomTypes);
    renderRooms(sortByPrice(allRooms));
  });

  // Nếu không dùng $.when, có thể gọi lần lượt:
  function loadAll() {
    $.ajax({
      url: 'URL', method: 'GET', data: { GetRooms: 'All' },
      success: function(data) {
        allRooms = data;
        renderRooms(sortByPrice(data));
      }
    });
    $.ajax({
      url: 'URL', method: 'GET', data: { RoomType: 'All' },
      success: function(data) {
        allRoomTypes = data;
        renderRoomTypes(data);
        // Thêm vào select filter
        $.each(data, function(i, t) {
          $('#roomTypeSelect').append(`<option value="${t.id}">${t.name}</option>`);
        });
      }
    });
  }
  loadAll();

  // Sort tăng dần theo giá
  function sortByPrice(rooms) {
    return [...rooms].sort((a, b) => a.price - b.price);
  }

  function renderRoomTypes(types) {
    $('#room-type-list').empty();
    $.each(types, function(i, t) {
      $('#room-type-list').append(`<li class="list-group-item">${t.name}</li>`);
    });
  }

  function renderRooms(rooms) {
    $('#room-list').empty();
    $.each(rooms, function(i, r) {
      $('#room-list').append(`
        <tr>
          <td>${r.roomNumber}</td>
          <td>${r.type}</td>
          <td>$${r.price}</td>
        </tr>
      `);
    });
  }

  // ════ b) Filter theo loại phòng ════
  $('#btnFilterType').click(function() {
    var type = $('#roomTypeSelect').val();
    if (!type || type === 'all') {
      renderRooms(sortByPrice(allRooms));
    } else {
      var filtered = allRooms.filter(r => r.type === type);
      renderRooms(sortByPrice(filtered));
    }
  });

  // ════ c) Thêm phòng mới với validation ════
  $('#btnAddRoom').click(function() {
    var isValid = true;

    // Reset
    $('.form-control').removeClass('is-invalid');
    $('.invalid-feedback').remove();

    var roomNumber = $('#roomNumber').val().trim();
    var price      = $('#roomPrice').val().trim();

    // Validate Room Number
    if (!roomNumber) {
      $('#roomNumber').addClass('is-invalid')
        .after('<div class="invalid-feedback">Please fill out this field.</div>');
      isValid = false;
    }

    // Validate Price: bắt buộc + phải là số
    if (!price) {
      $('#roomPrice').addClass('is-invalid')
        .after('<div class="invalid-feedback">Please fill out this field.</div>');
      isValid = false;
    } else if (isNaN(price) || Number(price) <= 0) {
      $('#roomPrice').addClass('is-invalid')
        .after('<div class="invalid-feedback">Price must be a valid number.</div>');
      isValid = false;
    }

    if (!isValid) return;

    // Gửi dữ liệu
    $.ajax({
      url: 'URL_TU_MOODLE', method: 'POST',
      data: { roomNumber, price },
      success: function(res) {
        alert(res.message || 'Thêm phòng thành công!');
        loadAll(); // reload lại danh sách
      }
    });
  });

  // Xóa lỗi khi gõ lại
  $(document).on('input', '.form-control', function() {
    $(this).removeClass('is-invalid');
    $(this).next('.invalid-feedback').remove();
  });

  // ════ d) Filter theo giá ════
  $('input[name="priceFilter"]').on('change', function() {
    var val = $(this).val(); // 'lt15' | '15to20' | 'gt20'
    var filtered;

    if (val === 'lt15') {
      filtered = allRooms.filter(r => r.price < 15);
    } else if (val === '15to20') {
      filtered = allRooms.filter(r => r.price >= 15 && r.price <= 20);
    } else {
      filtered = allRooms.filter(r => r.price > 20);
    }
    renderRooms(sortByPrice(filtered));
  });

});
